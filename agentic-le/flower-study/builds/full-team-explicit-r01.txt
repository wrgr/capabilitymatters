function fresh(){return {phase:'entry',which:0,draft:[['','',''],['']],first:[null,null],current:[null,null],shown:[false,false]};}
function escapeText(s){return String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));}
function button(text,action,secondary=false){return '<button type="button" data-action="'+action+'"'+(secondary?' class="secondary"':'')+'>'+text+'</button>';}
function listFor(w){return w===0?groups.slice(0,3):[groups[3]];}
function labelFor(g,v){if(v==='')return 'No response';if(v==='skip')return 'Skipped';if(v==='unsure')return 'Unsure';return g.options[Number(v)][0];}
function explanation(g,v){if(v==='')return 'No response — unassessed.';if(v==='skip')return 'Skipped — unassessed.';if(v==='unsure')return 'Uncertainty recorded; no competence judgment.
