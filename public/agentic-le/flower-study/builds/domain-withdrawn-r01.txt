const criteria=['Keep the observation within stated evidence; do not assert unknown motives.','Seek relevant missing context without presuming blame.','Address the uncertainty within the lead’s stated authority; keep the action conditional when needed.'];
function fresh(){return{attempts:[0,1].map(()=>({text:['','',''],status:'not started',history:[],examples:false,disagreed:false})),help:false,interruptions:0};}
function el(tag,text,parent=screen){const n=document.createElement(tag);if(text!==undefined)n.textContent=text;parent.appendChild(n);return n;}
function btn(text,fn,parent=screen){const b=el('button',text,parent);b.type='button';b.addEventListener('click',fn);return b;}
function title(text){screen.replaceChildren();const h=el('h2',text);h.id='screenTitle';h.tabIndex=-1;h.focus();}
function say(t){$('announce').textContent=t;}
function controls(){for(const id of ['pauseBtn','summaryBtn'])$(id).disabled=!state;}
function start(){view='start';title('A short practice invitation');el('p','Use fictional details only. Short fragments are enough. You may skip, revise, disagree, or stop. Several responses may fit.
